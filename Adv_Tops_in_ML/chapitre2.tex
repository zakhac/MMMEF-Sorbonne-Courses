\chapter{Classification Under Demographic Parity Constraint}

\section{Notations}

Considérons les notations suivantes pour ce chapitre :
\begin{itemize}
    \item Une observation est un triplet $(X, S, Y)$.
    \item $X \in \mathbb{R}^d$ représente les caractéristiques (features).
    \item $S \in \{-1, 1\}$ est l'attribut sensible (sensitive attribute).
    \item $Y \in \{0, 1\}$ est la sortie (label).
\end{itemize}

Un prédicteur (ou classifieur) $f$ est une fonction de $(X,S)$ telle que $f(X,S) \in \{0, 1\}$.

Nous considérons le risque de mauvaise classification (\textit{misclassification risk}) :
\begin{equation}
    R(f) = \mathbb{P}(f(X,S) \neq Y)
\end{equation}
comme mesure de performance.

Dans ce cas, le prédicteur optimal (ou Classifieur de Bayes) qui satisfait :
\begin{equation}
    f^* \in \argmin_{f} R(f)
\end{equation}
est défini par :
\begin{equation}
    f^*(X,S) = \indic\{\eta(X,S) \ge 1/2\}
\end{equation}
où $\eta(X,S) = \mathbb{P}(Y=1 | X,S)$.

\section{Définition de l'Équité (Fairness)}

Nous considérons la contrainte de Parité Démographique (Demographic Parity constraint ou DP-constraint).
Nous disons qu'un prédicteur $f$ satisfait la contrainte DP si :
\begin{equation}
    \forall s \in \{-1, 1\}, \quad \mathbb{P}(f(X,S)=1 | S=s) = \mathbb{P}(f(X,S)=1)
\end{equation}
Ceci équivaut aux propriétés suivantes :
\begin{align*}
    f(X,S) \perp \!\!\! \perp S \quad &\iff \mathbb{P}(f(X,S)=1 | S=1) = \mathbb{P}(f(X,S)=1 | S=-1) \\
    &\iff \sum_{s \in \{-1, 1\}} s \cdot \mathbb{P}(f(X,S)=1 | S=s) = 0
\end{align*}

\section{Classifieur Optimal sous Contrainte DP}

Nous considérons $f^*_{fair}$ défini comme :
\begin{equation}
    f^*_{fair} \in \argmin_{f \text{ satisfaisant DP}} R(f)
\end{equation}
Ceci est équivalent à résoudre le problème d'optimisation sous contrainte :
\begin{equation}
    f^*_{fair} \in \argmin_{\substack{f \\ \text{t.q. } \sum_{s \in \{-1, 1\}} s \cdot \mathbb{P}(f(X,S)=1 | S=s) = 0}} R(f)
\end{equation}

Le Lagrangien associé s'écrit comme suit :
\begin{equation}
    \mathcal{L}(f, \lambda) = R(f) + \lambda \sum_{s \in \{-1, 1\}} s \mathbb{P}(f(X,S)=1 | S=s)
\end{equation}
En introduisant la notation $\mathbb{P}(\cdot | S=s) := \mathbb{P}_{X|S=s}(\cdot)$, on a :
\begin{equation}
    \mathcal{L}(f, \lambda) = R(f) + \lambda \sum_{s \in \{-1, 1\}} s \mathbb{P}_{X|S=s}(f(X,S)=1)
\end{equation}

Pour résoudre le problème d'optimisation :
\begin{equation}
    f^*_{fair} \in \argmin_{f \text{ satisfaisant DP}} R(f)
\end{equation}
nous devons résoudre le problème d'optimisation suivant (problème primal) :
\begin{equation}
    \min_{f} \max_{\lambda} \mathcal{L}(f, \lambda)
\end{equation}

En général, nous résolvons le problème dual :
\begin{equation} \label{eq:dual_problem} \tag{$*$}
    \max_{\lambda} \min_{f} \mathcal{L}(f, \lambda)
\end{equation}
Nous avons toujours la propriété suivante (\textit{principe de dualité faible}) :
\begin{equation}
    \min_{f} \max_{\lambda} \mathcal{L}(f, \lambda) \ge \max_{\lambda} \min_{f} \mathcal{L}(f, \lambda)
\end{equation}
Nous résolvons le problème \eqref{eq:dual_problem} et montrons que :
\begin{equation}
    \min_{f} \max_{\lambda} \mathcal{L}(f, \lambda) = \max_{\lambda} \min_{f} \mathcal{L}(f, \lambda)
\end{equation}
C'est-à-dire qu'il n'y a pas de saut de dualité (\textit{no duality gap}, ou dualité forte).

\section{Résolution du Problème Dual}

Le but est d'obtenir une expression analytique (\textit{closed form expression}) du prédicteur optimal $f^*_{fair}$.

\subsection*{Première étape : Minimisation par rapport à $f$}

Pour tout $\lambda \in \mathbb{R}$ fixé, nous cherchons à résoudre :
\begin{equation}
    \min_{f} \mathcal{L}(f, \lambda)
\end{equation}
Rappelons l'expression du Lagrangien :
\begin{equation}
    \mathcal{L}(f, \lambda) = R(f) + \lambda \sum_{s \in \{-1, 1\}} s \mathbb{P}_{X|S=s}(f(X,S)=1)
\end{equation}
Le risque s'écrit comme une espérance :
\begin{align*}
    R(f) &= \mathbb{P}(f(X,S) \neq Y) \\
    &= \mathbb{E}[\indic\{f(X,S) \neq Y\}]
\end{align*}
En conditionnant par rapport à $Y$, on obtient :
\begin{align*}
    R(f) &= \mathbb{E}\left[ \indic\{f(X,S) \neq Y\} \indic\{Y=0\} \right] + \mathbb{E}\left[ \indic\{f(X,S) \neq Y\} \indic\{Y=1\} \right]
\end{align*}
Pour tout $y \in \{0, 1\}$, en utilisant la loi des espérances itérées par rapport à $(X,S)$ :
\begin{align*}
    \mathbb{E}\left[ \indic\{f(X,S) \neq y\} \indic\{Y=y\} \right] &= \mathbb{E}\left[ \mathbb{E}\left[ \indic\{f(X,S) \neq y\} \indic\{Y=y\} \big| X, S \right] \right] \\
    &= \mathbb{E}\left[ \indic\{f(X,S) \neq y\} \mathbb{E}\left[ \indic\{Y=y\} \big| X, S \right] \right] \\
    &= \mathbb{E}\left[ \indic\{f(X,S) \neq y\} \mathbb{P}(Y=y | X, S) \right]
\end{align*}
En notant $\eta(X,S) = \mathbb{P}(Y=1 | X,S)$, on a $\mathbb{P}(Y=0 | X,S) = 1 - \eta(X,S)$.
Ainsi, le risque devient :
\begin{align*}
    R(f) &= \mathbb{E}\left[ \indic\{f(X,S) \neq 0\} (1 - \eta(X,S)) \right] + \mathbb{E}\left[ \indic\{f(X,S) \neq 1\} \eta(X,S) \right]
\end{align*}
Puisque $f(X,S) \in \{0, 1\}$, nous avons les identités suivantes :
\begin{align*}
    \indic\{f(X,S) \neq 0\} &= \indic\{f(X,S) = 1\} = f(X,S) \\
    \indic\{f(X,S) \neq 1\} &= \indic\{f(X,S) = 0\} = 1 - f(X,S)
\end{align*}
En substituant ces identités dans l'expression du risque, on obtient :
\begin{align*}
    R(f) &= \mathbb{E}\left[ f(X,S) (1 - \eta(X,S)) \right] + \mathbb{E}\left[ (1 - f(X,S)) \eta(X,S) \right] \\
    &= \mathbb{E}\left[ f(X,S) (1 - 2\eta(X,S)) \right] + \mathbb{E}\left[ \eta(X,S) \right]
\end{align*}
De plus, par la loi des espérances itérées :
\begin{align*}
    \mathbb{E}[\eta(X,S)] &= \mathbb{E}[\mathbb{P}(Y=1 | X,S)] \\
    &= \mathbb{E}[\mathbb{E}[\indic\{Y=1\} | X,S]] \\
    &= \mathbb{E}[\indic\{Y=1\}] \\
    &= \mathbb{E}[Y]
\end{align*}
car $Y \in \{0,1\}$, donc $\mathbb{E}[Y|X,S] = \mathbb{P}(Y=1|X,S) = \eta(X,S)$.
Ainsi :
\begin{equation}
    R(f) = \mathbb{E}[Y] + \mathbb{E}\left[ f(X,S) (1 - 2\eta(X,S)) \right]
\end{equation}

Notons $\pi_s = \mathbb{P}(S=s)$. Nous pouvons décomposer l'espérance sur $S$ :
\begin{align*}
    R(f) &= \mathbb{E}[Y] + \sum_{s \in \{-1, 1\}} \mathbb{P}(S=s) \mathbb{E}_{X|S=s}\left[ f(X,S) (1 - 2\eta(X,S)) \right] \\
    &= \mathbb{E}[Y] + \sum_{s \in \{-1, 1\}} \pi_s \mathbb{E}_{X|S=s}\left[ f(X,S) (1 - 2\eta(X,S)) \right]
\end{align*}
Reprenons le Lagrangien :
\begin{align*}
    \mathcal{L}(f, \lambda) &= R(f) + \lambda \sum_{s \in \{-1, 1\}} s \mathbb{E}_{X|S=s}[f(X,S)] \\
    &= \mathbb{E}[Y] + \sum_{s \in \{-1, 1\}} \pi_s \mathbb{E}_{X|S=s}\left[ f(X,S) (1 - 2\eta(X,S)) \right] + \sum_{s \in \{-1, 1\}} \lambda s \mathbb{E}_{X|S=s}[f(X,S)] \\
    &= \mathbb{E}[Y] + \sum_{s \in \{-1, 1\}} \mathbb{E}_{X|S=s} \left[ \pi_s f(X,S) (1 - 2\eta(X,S)) + \lambda s f(X,S) \right] \\
    &= \mathbb{E}[Y] + \sum_{s \in \{-1, 1\}} \mathbb{E}_{X|S=s} \left[ \left( \pi_s (1 - 2\eta(X,S)) + \lambda s \right) f(X,S) \right]
\end{align*}
Ainsi, chercher $f^*_{\lambda} \in \argmin_{f} \mathcal{L}(f, \lambda)$ revient à minimiser point par point le terme sous l'espérance.
Puisque $f(X,S) \in \{0, 1\}$, pour minimiser une expression de la forme $A \cdot f(X,S)$, nous choisissons :
\begin{equation*}
    f(X,S) = 1 \iff A < 0
\end{equation*}
Ici, $A = \pi_s (1 - 2\eta(X,S)) + \lambda s$.
Nous avons donc :
\begin{align*}
    f^*_{\lambda}(X,S) = 1 &\iff \pi_s (1 - 2\eta(X,S)) + \lambda s < 0 \\
    &\iff 1 - 2\eta(X,S) < - \frac{\lambda s}{\pi_s} \\
    &\iff 2\eta(X,S) > 1 + \frac{\lambda s}{\pi_s} \\
    &\iff \eta(X,S) > \frac{1}{2} + \frac{\lambda s}{2\pi_s}
\end{align*}
Le prédicteur optimal pour un $\lambda$ fixé est donc :
\begin{equation}
    f^*_{\lambda}(X,S) = \indic\left\{ \eta(X,S) > \frac{1}{2} + \frac{\lambda s}{2\pi_s} \right\}
\end{equation}

\subsection*{Deuxième étape : Maximisation par rapport à $\lambda$}

Nous cherchons maintenant à résoudre :
\begin{equation}
    \max_{\lambda \in \mathbb{R}} \mathcal{L}(f^*_{\lambda}, \lambda)
\end{equation}
Soit $\lambda^*$ la solution :
\begin{equation}
    \lambda^* \in \argmax_{\lambda \in \mathbb{R}} \mathcal{L}(f^*_{\lambda}, \lambda)
\end{equation}
En réinjectant l'expression de $f^*_{\lambda}$ dans le Lagrangien, on obtient pour tout $\lambda \in \mathbb{R}$ :
\begin{align}
    \mathcal{L}(f^*_{\lambda}, \lambda) &= \mathbb{E}[Y] + \sum_{s \in \{-1, 1\}} \mathbb{E}_{X|S=s} \left[ \left( \pi_s (1 - 2\eta(X,S)) + \lambda s \right) f^*_{\lambda}(X,S) \right] \nonumber \\
    &= \mathbb{E}[Y] + \sum_{s \in \{-1, 1\}} \mathbb{E}_{X|S=s} \left[ \min\left( \pi_s (1 - 2\eta(X,S)) + \lambda s, 0 \right) \right]
\end{align}

Nous observons que :
\begin{equation*}
    \lambda^* \in \argmax_{\lambda \in \mathbb{R}} \sum_{s \in \{-1, 1\}} \mathbb{E}_{X|S=s} \left[ \min\left( \pi_s (1 - 2\eta(X,S)) + \lambda s, 0 \right) \right]
\end{equation*}
Ceci est équivalent à :
\begin{equation}
    \lambda^* \in \argmin_{\lambda \in \mathbb{R}} \sum_{s \in \{-1, 1\}} \mathbb{E}_{X|S=s} \left[ \max\left( \pi_s (2\eta(X,S) - 1) - \lambda s, 0 \right) \right]
\end{equation}
en utilisant l'identité $\max(A) = - \min(-A)$.

La fonction $\lambda \mapsto \max(\pi_s (2\eta(X,S) - 1) - \lambda s, 0)$ est convexe.
Par conséquent, la fonction objectif :
\begin{equation*}
    \lambda \mapsto \sum_{s \in \{-1, 1\}} \mathbb{E}_{X|S=s} \left[ \max\left( \pi_s (2\eta(X,S) - 1) - \lambda s, 0 \right) \right]
\end{equation*}
est convexe.

De plus, la fonction
\begin{equation}
    \mathcal{H}(\lambda) = \sum_{s \in \{-1, 1\}} \mathbb{E}_{X|S=s} \left[ \max\left( \pi_s (2\eta(X,S) - 1) - \lambda s, 0 \right) \right]
\end{equation}
est coercive, c'est-à-dire que $\lim_{|\lambda| \to +\infty} \mathcal{H}(\lambda) = +\infty$.

\begin{exemple}[Exercice : Coercivité et bornes sur $\lambda^*$]
    \textbf{Énoncé :}
    Montrer que la fonction $\lambda \mapsto \mathcal{H}(\lambda)$ est coercive et que le paramètre optimal vérifie $\lambda^* \in [-1, 1]$.

    \medskip
    \textbf{Correction :}
    \begin{enumerate}
        \item \textbf{Coercivité de $\mathcal{H}(\lambda)$}
        
        Rappelons que $\mathcal{H}(\lambda) = \sum_{s \in \{-1, 1\}} \mathbb{E}_{X|S=s} \left[ (\pi_s (2\eta(X,S) - 1) - \lambda s)_+ \right]$.
        
        Étudions le comportement quand $\lambda \to +\infty$ :
        \begin{itemize}
            \item Pour $s = 1$, le terme sous l'espérance est $(\pi_{1}(2\eta - 1) - \lambda)_+$. Puisque $2\eta - 1 \le 1$ et $\pi_1 \le 1$, on a $\pi_1(2\eta - 1) - \lambda \le 1 - \lambda$. Pour $\lambda > 1$, ce terme est négatif, donc la partie positive est nulle.
            \item Pour $s = -1$, le terme est $(\pi_{-1}(2\eta - 1) + \lambda)_+$. Pour $\lambda$ assez grand, $\pi_{-1}(2\eta - 1) + \lambda > 0$ presque sûrement. Le terme se comporte comme $\lambda$.
        \end{itemize}
        Ainsi, $\lim_{\lambda \to +\infty} \mathcal{H}(\lambda) = +\infty$. Un raisonnement symétrique s'applique pour $\lambda \to -\infty$ (le terme $s=1$ domine). La fonction est donc coercive.
        
        \item \textbf{Bornes sur $\lambda^*$}
        
        Montrons que pour tout $\lambda \ge 1$, $\mathcal{H}(\lambda) \ge \mathcal{H}(1)$ (et par symétrie pour $\lambda \le -1$).
        
        Soit $\lambda \ge 1$.
        \begin{itemize}
            \item Si $s=1$ : Comme vu précédemment, $\pi_1(2\eta - 1) - \lambda \le \pi_1 - \lambda \le 1 - 1 = 0$. Donc la contribution est nulle.
            \item Si $s=-1$ : Le terme est $(\pi_{-1}(2\eta - 1) + \lambda)_+$.
            Or $2\eta - 1 \ge -1$, donc $\pi_{-1}(2\eta - 1) \ge -\pi_{-1}$.
            Ainsi $\pi_{-1}(2\eta - 1) + \lambda \ge -\pi_{-1} + \lambda$.
            Comme $\lambda \ge 1$ et $\pi_{-1} \le 1$, ce terme est positif.
            La partie positive est donc simplement $\pi_{-1}(2\eta - 1) + \lambda$.
        \end{itemize}
        Ainsi, sur l'intervalle $[1, +\infty[$, $\mathcal{H}(\lambda) = \mathbb{E}_{X|S=-1}[\pi_{-1}(2\eta(X,S) - 1)] + \lambda$.
        C'est une fonction affine strictement croissante de $\lambda$.
        
        Soit maintenant $\lambda \le -1$.
        \begin{itemize}
            \item Si $s=-1$ : Le terme est $(\pi_{-1}(2\eta - 1) + \lambda)_+$. Comme $\pi_{-1}(2\eta - 1) \le \pi_{-1} \le 1$ et $\lambda \le -1$, la somme est négative. La contribution est nulle.
            \item Si $s=1$ : Le terme est $(\pi_{1}(2\eta - 1) - \lambda)_+$.
            Or $\pi_{1}(2\eta - 1) \ge -\pi_{1}$ et $-\lambda \ge 1$.
            Donc $\pi_{1}(2\eta - 1) - \lambda \ge 1 - \pi_{1} \ge 0$.
            La partie positive est donc $\pi_{1}(2\eta - 1) - \lambda$.
        \end{itemize}
        Ainsi, sur l'intervalle $]-\infty, -1]$, $\mathcal{H}(\lambda) = \mathbb{E}_{X|S=1}[\pi_{1}(2\eta(X,S) - 1)] - \lambda$.
        C'est une fonction affine strictement décroissante de $\lambda$.
        
        On en déduit que pour tout $|\lambda| > 1$, $\mathcal{H}(\lambda) > \min(\mathcal{H}(1), \mathcal{H}(-1))$.
        Par conséquent, le minimum global $\lambda^*$ doit se trouver dans l'intervalle $[-1, 1]$.
    \end{enumerate}
\end{exemple}

\begin{remarque}[Autre approche pour dériver le prédicteur équitable optimal]
    Nous avons établi que pour tout $\lambda \in \mathbb{R}$, le minimiseur du lagrangien est donné par :
    \[
    f_\lambda \in \argmin_f \mathcal{L}(f, \lambda) \quad \text{et} \quad f_\lambda(X,S) = \indic_{\left\{\eta(X,S) \ge \frac{1}{2} + \frac{\lambda s}{2\pi_s}\right\}}
    \]
    
    Nous cherchons maintenant $\lambda^*$ tel que $f_{\lambda^*}$ satisfasse la contrainte de Parité Démographique (DP). C'est-à-dire que nous devons trouver $\lambda^*$ tel que :
    \[
    \mathbb{P}(f_{\lambda^*}(X,S)=1 | S=1) = \mathbb{P}(f_{\lambda^*}(X,S)=1 | S=-1)
    \]
    
    Notons par $F_s(t)$ la fonction de queue (ou survie) de $\eta(X,S)$ sachant $S=s$ :
    \[
    F_s(t) = \mathbb{P}(\eta(X,S) > t | S=s)
    \]
    
    La condition $f_{\lambda^*}(X,s) = 1$ est équivalente à $\eta(X,s) \ge \frac{1}{2} + \frac{\lambda^* s}{2\pi_s}$.
    
    Si nous supposons que la distribution de $\eta(X,S)$ sachant $S=s$ est continue (pour éviter les problèmes aux bornes des indicatrices), alors :
    \[
    \mathbb{P}(f_{\lambda^*}(X,S)=1 | S=s) = F_s\left(\frac{1}{2} + \frac{\lambda^* s}{2\pi_s}\right)
    \]
    
    Nous devons donc résoudre l'équation suivante en $\lambda$ :
    \[
    F_1\left(\frac{1}{2} + \frac{\lambda}{2\pi_1}\right) = F_{-1}\left(\frac{1}{2} - \frac{\lambda}{2\pi_{-1}}\right)
    \]
    
    Remarquons que :
    \begin{itemize}
        \item Le terme de gauche, $\lambda \mapsto F_1\left(\frac{1}{2} + \frac{\lambda}{2\pi_1}\right)$, est décroissant (car $F_1$ est décroissante et $\lambda$ apparaît avec un signe $+$).
        \item Le terme de droite, $\lambda \mapsto F_{-1}\left(\frac{1}{2} - \frac{\lambda}{2\pi_{-1}}\right)$, est croissant (car $F_{-1}$ est décroissante et $\lambda$ apparaît avec un signe $-$).
    \end{itemize}
    
    Si l'on suppose de plus que les applications $t \mapsto F_s(t)$ sont continues, alors les deux membres de l'équation sont continus. L'existence et l'unicité de la solution (sous certaines conditions de support strictes) peuvent alors être étudiées via ces propriétés de monotonie, nous assurant de trouver le $\lambda^*$ optimal.
\end{remarque}

Ceci implique que le minimiseur $\lambda^*$ existe et $\lambda^* \in [-1, 1]$.

Pour terminer la dérivation de $f^*_{fair}$, nous devons considérer l'hypothèse suivante :
\begin{equation}
    t \mapsto \mathbb{P}_{X|S=s}(\eta(X,S) \le t) \quad \text{est continue.}
\end{equation}
Ceci implique que $\forall x \in \mathbb{R}, \mathbb{P}_{X|S=s}(\eta(X,S) = x) = 0$.

Sous cette hypothèse, nous avons que $\lambda \mapsto \mathcal{H}(\lambda)$ est différentiable (par rapport à $\lambda$).

Si nous considérons $h_s(\lambda) = \pi_s (2\eta(X,S) - 1) - \lambda s$, alors :
\begin{align*}
    \mathbb{E}_{X|S=s} \left[ \max(h_s(\lambda), 0) \right] &= \mathbb{E}_{X|S=s} \left[ h_s(\lambda) \indic\{h_s(\lambda) > 0\} \right] \\
    &\quad + \mathbb{E}_{X|S=s} \left[ 0 \cdot \indic\{h_s(\lambda) < 0\} \right] \\
    &\quad + \mathbb{E}_{X|S=s} \left[ h_s(\lambda) \indic\{h_s(\lambda) = 0\} \right]
\end{align*}
Or, $h_s(\lambda) = 0 \iff \pi_s (2\eta(X,S) - 1) - \lambda s = 0 \iff \eta(X,S) = \frac{1}{2} + \frac{\lambda s}{2\pi_s}$.
D'après notre hypothèse de continuité, la probabilité de cet événement est nulle.
Ainsi, le dernier terme est nul et le deuxième terme est nul car on multiplie par 0.
Pour le premier terme, $h_s(\lambda)$ est dérivable par rapport à $\lambda$ avec $\frac{\partial h_s(\lambda)}{\partial \lambda} = -s$.

On en déduit que $\lambda \mapsto \mathcal{H}(\lambda)$ est différentiable et sa dérivée est donnée par :
\begin{equation}
    \frac{\partial \mathcal{H}}{\partial \lambda}(\lambda) = \sum_{s \in \{-1, 1\}} \mathbb{E}_{X|S=s} \left[ \frac{\partial h_s(\lambda)}{\partial \lambda} \indic\{h_s(\lambda) > 0\} \right] = \sum_{s \in \{-1, 1\}} -s \cdot \mathbb{P}_{X|S=s}(h_s(\lambda) > 0)
\end{equation}

De plus, nous avons :
\begin{align*}
    \mathbb{E}_{X|S=s}[\indic\{h_s(\lambda) > 0\}] &= \mathbb{P}_{X|S=s}(h_s(\lambda) > 0) \\
    &= \mathbb{P}_{X|S=s}\left(\eta(X,S) > \frac{1}{2} + \frac{\lambda s}{2\pi_s}\right) \\
    &= \mathbb{P}_{X|S=s}(f^*_{\lambda}(X,S) = 1)
\end{align*}
Ainsi, pour tout $\lambda \in \mathbb{R}$ :
\begin{equation}
    \frac{\partial \mathcal{H}}{\partial \lambda}(\lambda) = - \sum_{s \in \{-1, 1\}} s \mathbb{P}_{X|S=s}(f^*_{\lambda}(X,S)=1)
\end{equation}
Puisque $\lambda^*$ est un minimum de $\mathcal{H}$ (fonction convexe différentiable et coercive sur $\mathbb{R}$), nous avons :
\begin{equation}
    \frac{\partial \mathcal{H}}{\partial \lambda}(\lambda^*) = 0 \iff \sum_{s \in \{-1, 1\}} s \mathbb{P}_{X|S=s}(f^*_{\lambda^*}(X,S)=1) = 0
\end{equation}
Ceci montre que le classifieur $f^*_{\lambda^*}$ satisfait la contrainte de Parité Démographique.
